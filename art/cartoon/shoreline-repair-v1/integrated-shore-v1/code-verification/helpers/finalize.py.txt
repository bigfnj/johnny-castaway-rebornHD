import hashlib
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[3]
OUT = Path(__file__).resolve().parent
CURRENT = OUT / 'full-canvas-current-catalog'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def rel(path):
    return path.relative_to(ROOT).as_posix()


sources = [
    'src/engine/art_style.c', 'src/engine/art_style.h', 'src/engine/graphics.c',
    'src/engine/island.c', 'tools/art_common.py', 'tools/art_pack.py',

    'docs/cartoon-footprint-contract.md', 'docs/cartoon-art.md',
    'docs/knowledge-base/cartoon-art-metadata.json',
    'docs/knowledge-base/cartoon-art-metadata.md',
    'docs/knowledge-base/cartoon-production-catalog.json',
    'docs/knowledge-base/cartoon-production-catalog.md',
]
reports = [OUT / 'full-canvas-smoke/report.json', OUT / 'full-canvas-regression/report.json',
           OUT / 'unix-flow/report.json', CURRENT / 'authoring-smoke.json',
           CURRENT / 'authoring-regression.json', CURRENT / 'authoring-mutations.json']
for path in reports:
    record = json.loads(path.read_bytes())
    assert record['status'] == 'PASS', path
    for name, expected in record.get('source_sha256', {}).items():
        assert sha(ROOT / name) == expected, name
for phase in ('smoke', 'regression'):
    record = json.loads((CURRENT / ('authoring-' + phase + '.json')).read_bytes())
    for run in record['runs']:
        assert sha(CURRENT / run['log']) == run['log_sha256'], run['log']
mutations = json.loads((OUT / 'full-canvas-regression/report.json').read_bytes())['mutations']
for record in mutations:
    assert record['exit_code'] == 2, record
    assert record['output'].count('FAIL ') == 1, record
    assert 'FAIL ' + record['case'] + ':' in record['output'], record
    assert record['build']['rebuilt_timestamp_advanced'], record

commands = [
    [sys.executable, '-B', 'tools/art_review_metadata.py', '--check'],
    [sys.executable, '-B', 'tools/art_production_catalog.py', '--check'],
    ['powershell.exe', '-NoProfile', '-Command',
     "$errors = $null; $tokens = $null; [System.Management.Automation.Language.Parser]::ParseFile('tests/Test-ArtMutations.ps1', [ref]$tokens, [ref]$errors) | Out-Null; if ($errors.Count) { $errors | Out-String; exit 1 }; Write-Output 'PASS PowerShell 5.1 parser: tests/Test-ArtMutations.ps1'"],
    ['git', 'diff', '--check'],
]
checks = []
for i, command in enumerate(commands):
    result = subprocess.run(command, cwd=ROOT, capture_output=True, text=True)
    log = OUT / ('final-check-' + str(i) + '.log')
    log.write_text(result.stdout + result.stderr, encoding='utf-8')
    assert result.returncode == 0, log.read_text()
    portable = command.copy()
    if portable[0] == sys.executable:
        portable[0] = 'TOOLBOX_PYTHON'
    checks.append({'command': portable, 'exit_code': result.returncode,
                   'log': rel(log), 'sha256': sha(log)})

archive = ROOT / 'assets/scrantic_data.zip'
